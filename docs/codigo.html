# Devuelve una función de Python completa: desde «def nombre(» hasta la siguiente línea sin sangría.
funcion_py <- function(nombre) {
  inicio <- grep(paste0("^def ", nombre, "\\("), ENTRENAR)
  resto <- seq(inicio + 1, length(ENTRENAR))
  fin <- resto[grepl("^\\S", ENTRENAR[resto])][1] - 1
  while (!nzchar(trimws(ENTRENAR[fin]))) fin <- fin - 1
  ENTRENAR[inicio:fin]
}

# Devuelve las líneas entre dos marcas de texto (incluidas).
tramo_py <- function(desde, hasta) {
  inicio <- grep(desde, ENTRENAR, fixed = TRUE)[1]
  fin <- inicio - 1 + grep(hasta, ENTRENAR[inicio:length(ENTRENAR)], fixed = TRUE)[1]
  ENTRENAR[inicio:fin]
}

9.1 Cómo se dividieron los datos

La cohorte se reparte en siete grupos del mismo tamaño con StratifiedGroupKFold. Estratificado significa que cada grupo conserva la proporción de personas fuera de la ruta formal (21 %); agrupado significa que todas las personas de una vivienda caen en el mismo grupo. De los siete grupos, cinco forman el entrenamiento (≈ 71 %), uno la validación (≈ 14 %) y uno la prueba (≈ 14 %). La función se detiene si alguna vivienda aparece en dos particiones.

Ver código Python 
def group_ids(df: pd.DataFrame) -> np.ndarray:
    # Separa viviendas completas, incluso cuando contienen más de un hogar.
    # Incluye municipio y clase: COD_ENCUESTAS no se presume global.
    return (
        df[DWELL_KEY].fillna(-1).astype("int64").astype("string")
        .agg("|".join, axis=1).to_numpy()
    )

def split_three(x, y, groups):
    splitter = StratifiedGroupKFold(n_splits=7, shuffle=True, random_state=SEED)
    folds = [test for _, test in splitter.split(x, y, groups)]
    train = np.concatenate(folds[2:])
    val, test = folds[1], folds[0]
    if (
        set(groups[train]) & set(groups[val]) or
        set(groups[train]) & set(groups[test]) or
        set(groups[val]) & set(groups[test])
    ):
        raise AssertionError("Una vivienda aparece en más de una partición")
    return train, val, test

9.2 Preprocesamiento dentro del pipeline

La imputación y la codificación forman parte de cada pipeline. Así se aprenden solo con los datos de ajuste de cada pliegue y no con validación ni prueba, lo que evita fuga de información.

Ver código Python 
def encoded_preprocessor(age_as_category=False, numeric=NUMERIC):
    cats = CATEGORICAL + (["P_EDADR"] if age_as_category else [])
    nums = [column for column in numeric if column not in cats]
    return ColumnTransformer([
        ("cat", OneHotEncoder(handle_unknown="ignore", min_frequency=50), cats),
        ("num", Pipeline([
            ("impute", SimpleImputer(strategy="median", add_indicator=True)),
            ("scale", StandardScaler()),
        ]), nums),
    ])

def hist_preprocessor(numeric=NUMERIC):
    return ColumnTransformer([
        ("cat", OrdinalEncoder(
            handle_unknown="use_encoded_value", unknown_value=-1,
        ), CATEGORICAL),
        ("num", SimpleImputer(strategy="median", add_indicator=True), numeric),
    ], sparse_threshold=0)

9.3 Candidatos, búsqueda de hiperparámetros y elección

Cada familia se ajusta con GridSearchCV y validación cruzada de tres pliegues, también agrupada por vivienda y solo dentro del entrenamiento. La métrica de búsqueda es la precisión promedio (average_precision). Después, cada ganador se mide en validación; el modelo con mayor AP de validación es el elegido.

9.4 Métricas y umbral

metricas calcula todas las medidas para una clase positiva SIN_ACCESO = 1. optimize_threshold busca en validación el umbral que maximiza F1; ese valor se fija antes de abrir la prueba.

Ver código Python 
def metricas(y, score, threshold=0.5):
    pred = (score >= threshold).astype(int)
    tn, fp, fn, tp = confusion_matrix(y, pred, labels=[0, 1]).ravel()
    return {
        "threshold": float(threshold),
        "roc_auc": float(roc_auc_score(y, score)),
        "pr_auc": float(average_precision_score(y, score)),
        "brier": float(brier_score_loss(y, score)),
        "f1": float(f1_score(y, pred, zero_division=0)),
        "precision": float(precision_score(y, pred, zero_division=0)),
        "recall": float(recall_score(y, pred, zero_division=0)),
        "specificity": float(tn / (tn + fp)),
        "confusion": {"tn": int(tn), "fp": int(fp), "fn": int(fn), "tp": int(tp)},
    }

def optimize_threshold(y, score):
    precision, recall, threshold = precision_recall_curve(y, score)
    f1 = 2 * precision[:-1] * recall[:-1] / np.maximum(
        precision[:-1] + recall[:-1], 1e-12
    )
    best = int(np.argmax(f1))
    return float(threshold[best]), float(f1[best])

g_brecha <- function() {
  d <- leer("auditoria_brecha_2026.csv")
  bandas <- c("0-14" = "0 a 14", "15-59" = "15 a 59", "60+" = "60 o más")
  p <- plotly::plot_ly(height = 400)
  for (grupo in c("Con dificultad", "Sin dificultad")) {
    parte <- d[d$dificultad == grupo, ]
    p <- plotly::add_trace(p, x = factor(bandas[parte$edad], levels = bandas), y = parte$sensibilidad,
                           type = "scatter", mode = "lines+markers", name = grupo,
                           line = list(color = if (grupo == "Con dificultad") TK$grupo_b else TK$grupo_a, width = 2.5),
                           marker = list(size = 9),
                           error_y = list(type = "data", symmetric = FALSE, array = parte$superior - parte$sensibilidad,
                                          arrayminus = parte$sensibilidad - parte$inferior, thickness = 1.5, width = 5),
                           customdata = paste0(grupo, ", ", bandas[parte$edad], " años<br>Sensibilidad: ", pct_es(100 * parte$sensibilidad),
                                               "<br>Detectados: ", num_es(parte$detectados), " de ", num_es(parte$positivos),
                                               "<br>Intervalo Wilson 95 %: ", pct_es(100 * parte$inferior), " a ", pct_es(100 * parte$superior)),
                           hovertemplate = "%{customdata}<extra></extra>")
  }
  estilo_libro(p, "Sensibilidad por edad y dificultad funcional declarada", leyenda = TRUE,
               margen = list(l = 10, r = 20, t = 84, b = 10)) |>
    plotly::layout(xaxis = eje("Edad en años", gridcolor = "rgba(0,0,0,0)"),
                   yaxis = eje("Sensibilidad entre quienes están fuera de la ruta", range = c(0, 1), tickformat = ".0%"))
}
